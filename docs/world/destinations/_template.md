    enum:
      - civilian
      - military
      - black-market
      - research

    type: object
    properties:
      trader:
        type: boolean
        description: buy/sell commodities
      mission_board:
        type: boolean
        description: take and hand in missions
      ship_repair:
        type: boolean
        description: repair hull damage
      fuel:
        type: boolean
        description: refuel jump drive
      ship_dealer:
        type: boolean
        description: buy or trade ships
    required:
      - trader
      - mission_board
      - ship_repair
      - fuel
      - ship_dealer

    type: object
    description: only include an entry if the corresponding amenity is true

      trader:

      ship_dealer:
        type: string
        description: display name of the ship dealer NPC

    type: array
    items:
      type: string

      economy tags that influence commodity pricing at this destination.

  danger_level:
    enum:
      - none
      - low
      - medium
      - high
      - extreme
  tags:
    type: array
    items:
      type: string

  - id
  - name
  - system
  - location_type
  - type
  - amenities
